% ------ SLIDE 3: BREACHES ----------------------------------------
\begin{frame}{Biometric Breaches --- The Scale of the Problem}
\footnotesize
\begin{itemize}
  \item \textbf{Aadhaar (India, 2018):} 1.1~billion biometric records exposed
        in a single breach of the centralised UIDAI database
  \item \textbf{US Office of Personnel Management (2015):}
        5.6~million fingerprint records stolen; used for intelligence operations
  \item \textbf{BioStar~2 (2019):}
        28~million biometric records including fingerprints and facial templates
        left exposed on an unprotected server
  \item \textbf{India's FASTag (2023):}
        facial and fingerprint data of millions of vehicle owners leaked
\end{itemize}
\medskip
\lbl{Why biometrics are uniquely dangerous:}
\begin{itemize}
  \item A stolen password can be changed in seconds
  \item A stolen fingerprint \bad{cannot be changed --- ever}
  \item An attacker with your biometric template has lifelong leverage
  \item Every new system you enroll in is instantly compromised
\end{itemize}
\end{frame}

% ------ SLIDE 4: WHY TEMPLATES ARE WEAK -------------------------
\begin{frame}{Why Stored Templates Are the Weak Link}
\footnotesize
\lbl{Raw template storage (status quo):}
\begin{itemize}
  \item Biometric scanner extracts a feature vector (minutiae, iris code)
  \item This raw vector is stored in a central database
  \item Attacker who steals the database can:
    \begin{itemize}
      \item Replay the template to spoof any system that enrolled from it
      \item Cross-match across databases to de-anonymize users
      \item Reconstruct the physical trait using generative models (GAN attacks)
    \end{itemize}
\end{itemize}
\medskip
\lbl{The single-server problem:}
\begin{itemize}
  \item Even protected templates stored on one server create one target
  \item A single database breach exposes the complete vault to offline attack
  \item Offline polynomial reconstruction:
        $\approx 6.8 \times 10^8$ attempts at $10^6$/sec
        $= $ \bad{11 minutes} (detailed in Security section)
  \item Distributed storage eliminates this single point of failure
\end{itemize}
\end{frame}

% ------ SLIDE 5: ISO/IEC 24745 ----------------------------------------
\begin{frame}{ISO/IEC 24745 --- What a Compliant System Must Guarantee}
\footnotesize
The international standard for biometric information protection
defines three \textbf{mandatory} properties:
\medskip
\begin{itemize}
  \item[\tick] \textbf{Irreversibility}\\
        {\scriptsize
        It must be computationally infeasible to reconstruct the original
        biometric trait from the stored template, even with full knowledge
        of the protection scheme and its parameters.\\
        \emph{Why it matters:} prevents reconstruction attacks;
        recovering the fingerprint from a stolen vault must require
        solving ECDLP ($2^{128}$ operations).}
  \item[\tick] \textbf{Unlinkability}\\
        {\scriptsize
        Templates created from the same biometric for different
        applications must be mathematically unrelated.
        An adversary who compromises one application's database must
        learn nothing about the same user's template in another.\\
        \emph{Why it matters:} prevents cross-application de-anonymization.}
  \item[\tick] \textbf{Revocability}\\
        {\scriptsize
        If a template is compromised, it must be possible to cancel it
        and issue a new, independent template from the \emph{same} biometric.\\
        \emph{Why it matters:} enables damage control after a breach.}
\end{itemize}
\medskip
{\scriptsize\color{DGrey}
No prior multimodal fuzzy vault scheme satisfies all three properties
while also providing a formal zero-penalty multimodal fusion guarantee.
This thesis does.}
\end{frame}

% ------ SLIDE 6: ORIGINAL FV ----------------------------------------
\begin{frame}{The Original Fuzzy Vault (Juels \& Sudan, 2002)}
\footnotesize
\lbl{Core idea:}
\begin{itemize}
  \item Bind a secret $s$ to a noisy, unordered set $A$
        (e.g., fingerprint minutiae coordinates)
  \item Encode $s$ as a degree-$k$ polynomial $p(x)$ over GF($q$)
  \item Evaluate $p$ at each element of $A$
        $\to$ genuine points $\{(a_i, p(a_i))\}$
  \item Add hundreds of \textbf{chaff points} $(c_j, r_j)$
        where $r_j \neq p(c_j)$ --- these are fake, indistinguishable
  \item \textbf{Vault} = shuffle of genuine + chaff
\end{itemize}
\medskip
\lbl{Unlocking:}
\begin{itemize}
  \item Query features $B$ are matched against vault x-coordinates
  \item If $|A \cap B| \geq k{+}1$ genuine matches found:
        Lagrange interpolation reconstructs $p(x)$ and recovers $s$
  \item Chaff points produce garbage polynomials --- easily detected
\end{itemize}
\medskip
\lbl{Security intuition:}
\begin{itemize}
  \item Without knowing $A$, attacker must try all $\binom{N}{k+1}$ subsets
  \item With $N$ total points and most being chaff:
        brute-force is computationally infeasible
  \item Weakness: GF($2^{16}$) gives only $2^8$ security --- too weak
\end{itemize}
\end{frame}

% ------ SLIDE 7: ECC FV ENROLLMENT ----------------------------------------
\begin{frame}{ECC Fuzzy Vault --- Enrollment (Maurya et al., 2025)}
\footnotesize
\lbl{Why ECC?  GF($2^{16}$) gives only $2^8$ security.
ECC Brainpool P256r1 gives $2^{128}$.}
\medskip
\lbl{Enrollment steps:}
\begin{enumerate}\setlength\itemsep{3pt}
  \item \textbf{Quantize} minutia $(x_i,y_i,\theta_i)$
        by factor $f$: each dimension modulo 64
        $\to$ 18-bit integer $q_i$
  \item \textbf{EC Map:} $P_i = q_i \cdot G$ on Brainpool P256r1.
        Recovering $q_i$ from $P_i.x$ requires solving ECDLP:
        $\approx 2^{128}$ operations
  \item \textbf{Polynomial:}
        degree-$k$ poly $p(x)$ over GF($p$), coefficients in $[10,63]$.
        Secret $s = \bigoplus \mathrm{bin}_6(a_j) \bmod n$;
        public key $S = s \cdot G$
  \item \textbf{Genuine points:}
        $\mathcal{G} = \{(P_i.x,\;p(P_i.x))\}_{i=1}^{g}$
  \item \textbf{Chaff:} $5g$ random pairs $(c_j.x, r_j)$ with
        $r_j \neq p(c_j.x)$
  \item \textbf{Vault:}
        $\mathcal{V} = \mathrm{shuffle}(\mathcal{G} \cup \mathcal{C})$;
        stored with public key $S$
\end{enumerate}
\end{frame}

% ------ SLIDE 8: ECC FV VERIFICATION ----------------------------------------
\begin{frame}{ECC Fuzzy Vault --- Verification}
\footnotesize
\lbl{Given a query fingerprint, unlock the vault:}
\begin{enumerate}\setlength\itemsep{3pt}
  \item Quantize query minutiae $\mathcal{M}'$ to $\mathcal{Q}'$;
        compute $\{P'_i = q'_i \cdot G\}$
  \item Match query EC points to vault x-coordinates:
        $\mathcal{X}_m = \{(x,y) \in \mathcal{V} : x \in \{P'_i.x\}\}$
  \item If $|\mathcal{X}_m| \geq k{+}1$:
        try each $(k{+}1)$-subset via Lagrange interpolation,
        recover candidate $\hat{s}$
  \item Verify: accept if $\hat{s} \cdot G = S$
\end{enumerate}

% ------ SLIDE 9: IRIS PIPELINE ----------------------------------------
\begin{frame}{Iris Biometric Pipeline}
\footnotesize
\lbl{Why iris?
Daugman (2004) showed iris has the widest intra/inter-class separation
of any biometric --- FHD near 0.10--0.15 (genuine) vs 0.46 (impostor).}
\medskip
\lbl{Processing pipeline:}
\begin{enumerate}\setlength\itemsep{2pt}
  \item \textbf{Capture:} CASIA-Iris-Interval, $320\times280$ px,
        near-infrared illumination
  \item \textbf{Segment:} Detect pupil (intensity thresholding) and
        outer iris (radial gradient).
        Mask pupil, specular reflections, eyelids
  \item \textbf{Normalize:} Daugman rubber-sheet model
        $\to$ $64\times512$ rectangular strip
  \item \textbf{Encode:} CLAHE enhancement,
        1-D Log-Gabor filters (18 and 36 px wavelengths)
        $\to$ \textbf{131,072-bit iris code} (phase-quantized binary)
  \item \textbf{Match:} Fractional Hamming Distance (FHD) over valid bits,
        $\pm32$ column shifts for rotational tolerance
\end{enumerate}
\medskip
\lbl{Key property for fusion:}
\begin{itemize}
  \item Genuine FHD $\approx 0.10$--$0.15$ (low = similar)
  \item Impostor FHD $\approx 0.46$ (close to random = 0.5)
  \item Wide separation enables reliable binary key extraction
\end{itemize}
\end{frame}

% ------ SLIDE 10: IRIS FUZZY COMMITMENT ----------------------------------------
\begin{frame}{Iris Fuzzy Commitment --- Key Extraction}
\footnotesize
\lbl{Goal: extract a stable, secret binary key from a noisy iris code.}
\medskip
\lbl{Enrollment:}
\begin{enumerate}\setlength\itemsep{2pt}
  \item Partition 131,072-bit code into $n_b$ blocks using
        \textbf{bit interleaving}:
        bit $j$ goes to block $(j \bmod n_b)$.
        This \emph{spreads} spatially correlated noise (eyelid occlusion)
        across \emph{all} blocks instead of wiping out consecutive ones
  \item Generate random $n_b$-bit key $\mathbf{k}$
  \item Helper data: $\mathbf{h} = \mathbf{c} \oplus \mathbf{w}$
        (where $\mathbf{w}$ repeats $\mathbf{k}$ over all positions)
  \item Store $(\mathbf{h},\; H(\mathbf{k}))$ --- \emph{not} $\mathbf{k}$ itself
\end{enumerate}
\medskip
\lbl{Recovery (at query time):}
\begin{enumerate}\setlength\itemsep{2pt}
  \item For each rotation shift $\delta \in [-32, +32]$:
        compute $\tilde{\mathbf{c}} = \mathrm{rotate}(\mathbf{c}',\delta)$,
        then $\tilde{\mathbf{w}} = \tilde{\mathbf{c}} \oplus \mathbf{h}$
  \item For each block: \textbf{majority vote} over block's valid bits
        $\to \hat{k}_i$
  \item At $B_s=255$: capacity = 127 errors/block ($\approx49.8\%$);
        genuine FHD $\approx0.20$ ($\approx51$ errors/block) --- majority vote succeeds
  \item Accept if $H(\hat{\mathbf{k}}) = H(\mathbf{k})$
\end{enumerate}
\medskip
\begin{center}
\begin{tabular}{@{}lccc@{}}
  \toprule
  Block size $B_s$ & Key bits $n_b$ & GAR & Impostor FAR\\
  \midrule
  255 & 514 & ${\approx}62\%$ & ${\approx}0.10\%$\\
  \bottomrule
\end{tabular}
\end{center}
\end{frame}

% ------ SLIDE 11: ARCHITECTURE A ----------------------------------------
\begin{frame}{Existing Multimodal Approach --- Architecture A (AND-Fusion)}
\footnotesize
\lbl{What it does:}
\begin{itemize}
  \item Include genuine points from \emph{both} fingerprint
        and iris in the same vault
  \item Polynomial reconstruction requires $\geq k{+}1$ matches
        drawn from \emph{either} modality
  \item But in practice: the \textbf{AND constraint} applies ---
        both modalities must contribute enough points
\end{itemize}
\medskip
\lbl{The gating penalty:}
\[\GAR_{\mathrm{AND}}
    = \GAR_{\mathrm{FP}} \times \GAR_{\mathrm{Iris}}
    \leq \GAR_{\mathrm{FP}}\]
Our iris GAR $\approx 62\%$.
So \bad{38\% of genuine users are blocked by iris alone},
even when their fingerprint is perfect.
\medskip
\lbl{EER results for Architecture A vs Unimodal:}

% ------ SLIDE 12: ARCH B, C, D ----------------------------------------
\begin{frame}{Existing Multimodal Approaches --- Architectures B, C, D}
\footnotesize
Three alternative tight-coupling strategies all make the iris a
\textbf{mandatory decryption key}:
\medskip
\lbl{Architecture B --- Polynomial Blinding:}
\begin{itemize}
  \item XOR-mask each polynomial coefficient with an iris-derived key
  \item Blind polynomial $\to$ vault; recover key $\to$ unmask $\to$ decrypt
  \item Without iris key: polynomial is permanently masked.
        \bad{Vault is useless.}
\end{itemize}
\medskip
\lbl{Architecture C --- Dense Chaff Seeding:}
\begin{itemize}
  \item Iris key seeds a PRNG that injects $20\times$ extra chaff
  \item Without iris key: search space inflates $20\times$
        $\to$ computationally infeasible to reconstruct
  \item \bad{User is rejected even if fingerprint matches perfectly.}
\end{itemize}
\medskip
\lbl{Architecture D --- AES-256-GCM Vault Encryption:}
\begin{itemize}
  \item Entire vault is AES-encrypted with the iris-derived key
  \item Without iris key: vault is a block of ciphertext.
        \bad{Cannot even begin to decode.}
\end{itemize}
\end{frame}

% ------ SLIDE 14: ARCH B/C/D FLOOR ----------------------------------------
\begin{frame}{Why the 19\% Floor Is Unfixable}
\footnotesize
\lbl{The mathematics:}
\begin{itemize}
  \item Architectures B, C, D require iris key for vault decoding
  \item Iris key recovery GAR $\approx 62\%$
  \item Therefore: FRR $\geq 1 - 0.62 = 38\%$ by construction,
        \emph{regardless of fingerprint quality, threshold, or parameters}
\end{itemize}
\medskip
\lbl{EER calculation:}
\[\EER = \frac{\FRR + \FAR}{2} \approx \frac{38\% + 0\%}{2} = 19\%\]
\begin{itemize}
  \item This floor is \bad{structural} ---
        it cannot be removed without changing the architecture
  \item Any tuning that raises FAR to meet FRR at 19\% confirms the floor
  \item The \emph{only} solution: decouple iris from the decoding path
\end{itemize}
\medskip

% ------ SLIDE 16: CORE INSIGHT ----------------------------------------
\begin{frame}{The Core Insight --- Iris as Optional Booster}
\footnotesize
\lbl{The problem with all prior work:}\\
They asked: \emph{"How do we force iris into the decoding path?"}\\
This created the mandatory gate and the 19\% floor.
\medskip
\lbl{The right question:}\\
\emph{"How can iris help when it works, without hurting when it does not?"}
\medskip
\lbl{Our answer --- Iris Bonus Points:}
\begin{itemize}
  \item When iris succeeds ($\sim$62\% of genuine queries):
        derive $K$ virtual polynomial evaluation points from the iris key
  \item Inject these as \textbf{guaranteed genuine points} into the vault's
        candidate set
  \item Smart decoder fixes all $K$ bonus points,
        reducing the search space by $\approx 5.5\times$
  \item More genuine points in the candidate set
        $\to$ higher chance of finding $k{+}1$ matches
        $\to$ lower EER
  \item When iris fails ($\sim$38\%):
        \textbf{nothing changes} --- fall back to unimodal exactly
\end{itemize}
\medskip
\lbl{The consequence:}\\
It is \emph{mathematically impossible} for \IBFV{} to perform
worse than unimodal. Formally proven in Theorem 1.
\end{frame}

% ------ SLIDE 17: IBFV ENROLLMENT ----------------------------------------
\begin{frame}{IBFV Enrollment}
\footnotesize
\lbl{After standard ECC vault enrollment, three additional steps:}
\begin{enumerate}\setlength\itemsep{2pt}
  \item Extract iris key $\mathbf{k}$; compute $\kappa = H(\mathbf{k})$
  \item Derive $K$ virtual x-coordinates via PRF (SHA-256):
        $v_j = \mathrm{SHA\text{-}256}(\texttt{"ibfv\_vpt:"}\|j\|\kappa) \bmod p$
  \item Compute bonus genuine points:
        $\mathcal{G}_\mathrm{Iris} = \{(v_j,\;p(v_j))\}_{j=0}^{K-1}$
  \item Final vault:
        $\mathcal{V} = \mathrm{shuffle}(\mathcal{G}_\mathrm{FP}
        \cup \mathcal{G}_\mathrm{Iris} \cup \mathcal{C})$
\end{enumerate}
{\scriptsize Without $\kappa$, the $K$ x-coordinates are
computationally indistinguishable from chaff (SHA-256 preimage resistance).}
\end{frame}

% ------ SLIDE 18: IBFV VERIFY PATH A ----------------------------------------
\begin{frame}{IBFV Verification --- Path A (Iris Succeeds, $\sim$62\%)}
\footnotesize
\lbl{When iris key recovery succeeds:}
\begin{enumerate}\setlength\itemsep{2pt}
  \item Recover iris key $\hat{\mathbf{k}}$
        $\to$ derive $v_0, \ldots, v_{K-1}$ (same PRF as enrollment)
  \item Locate bonus matches in vault:

% ------ SLIDE 23: SHAMIR ALGORITHM ----------------------------------------
\begin{frame}{Shamir Secret Sharing --- The Algorithm}
\footnotesize
\lbl{Setup: $(t,n)$-threshold sharing over GF(257)}
\begin{itemize}
  \item Applied \textbf{independently to each byte} $b_i$ of the
        serialized vault $\mathbf{b}$
  \item Why GF(257)? Smallest prime field containing all 256 byte
        values (0--255). No truncation or modular artefacts.
\end{itemize}
\medskip
\lbl{Splitting (enrollment):}
\begin{enumerate}\setlength\itemsep{2pt}
  \item For each byte $b_i$:
        sample random polynomial $f_i(x)$ of degree $t{-}1$ over GF(257)
        with $f_i(0) = b_i$
  \item Compute $n$ shares: $s_{j,i} = f_i(j) \bmod 257$,
        for $j = 1, \ldots, n$
  \item Share $j$ collects $\{s_{j,0}, s_{j,1}, \ldots, s_{j,L-1}\}$
        for all $L$ vault bytes
\end{enumerate}
\medskip
\lbl{Reconstruction (verification):}
\begin{enumerate}\setlength\itemsep{2pt}
  \item Collect any $t$ shares
  \item For each byte position $i$:
        Lagrange interpolation at $x=0$ over GF(257)
        $\to$ exact $\hat{b}_i = b_i$
\end{enumerate}
\medskip
\lbl{$(t,n)$ configurations used:}
\begin{center}
\begin{tabular}{@{}ccccc@{}}
  \toprule
  $(t,n)$ & Fault tolerance & Split (ms) & Recon (ms) & IPFS storage\\
  \midrule
  $(2,3)$ & 1 node & 0.3 & 0.2 & 99\,KB\\
  $(3,5)$ & 2 nodes & 0.7 & 0.2 & 166\,KB\\
  $(4,7)$ & 3 nodes & 1.3 & 0.2 & 232\,KB\\
  \bottomrule
\end{tabular}
\end{center}
\end{frame}

% ------ SLIDE 25: IPFS ----------------------------------------
\begin{frame}{IPFS --- Distributed Content-Addressed Storage}
\footnotesize
\lbl{What is IPFS?}
\begin{itemize}
  \item A peer-to-peer storage network where each file is identified
        by its \textbf{Content Identifier (CID)},
        computed as the SHA-256 hash of its content
  \item Retrieving by CID = verifying integrity automatically:
        any tampering changes the CID, making the share undetectable as modified
  \item No single server hosts the data ---
        any node that \emph{pins} a CID serves it
\end{itemize}
\medskip
\lbl{Why IPFS for vault shares:}
\begin{itemize}
  \item \textbf{Integrity:} CID is a cryptographic commitment ---
        tampered share has a different CID, lookup fails
  \item \textbf{Decentralization:} shares spread across $n$ IPFS nodes;
        no single node is a single point of failure
  \item \textbf{Content addressing:} the blockchain stores
        only 0.5\,KB of CID pointers ---
        not the vault itself (16.6\,KB)
  \item \textbf{Retrievability:} any node that has pinned the share
        can respond to retrieval; no DNS, no IP dependency
\end{itemize}
\medskip
\lbl{D-IBFV usage:}
Each of the $n$ Shamir shares is uploaded as a separate IPFS object.
The $n$ CIDs are anchored on-chain.
Verification: fetch any $t$ shares using their CIDs.
\end{frame}

% ------ B8: IPFS INTERNALS ----------------------------------------
\begin{frame}{B8: IPFS Internals --- Merkle DAG and CIDs}
\footnotesize
\lbl{How IPFS stores and retrieves data:}
\begin{itemize}
  \item Data is split into blocks; each block is hashed (SHA-256)
  \item Blocks are organized into a \textbf{Merkle Directed Acyclic Graph (DAG)}:
        parent node hash depends on all child hashes
  \item \textbf{Content Identifier (CID):}
        a multihash encoding the hash algorithm + hash of the root block.
        CIDv1 = base32(multicodec + multihash)
  \item Retrieving a CID: broadcast to network
        $\to$ any node that has pinned it responds
        $\to$ hash is re-verified locally on receipt
\end{itemize}
\medskip
\lbl{Pinning:}
\begin{itemize}
  \item By default IPFS garbage-collects unpinned data
  \item Pinning a CID means a node commits to storing it permanently
  \item In D-IBFV: enrollment node pins all $n$ shares;
        additional nodes can be incentivized (Filecoin)
\end{itemize}
\medskip
\lbl{Why integrity is automatic:}\\
Tampered share $\to$ different SHA-256 $\to$ different CID
$\to$ blockchain lookup fails $\to$ retrieval is rejected.
No separate integrity check needed.
\end{frame}

% ------ B9: FABRIC CONSENSUS ----------------------------------------
\begin{frame}{B9: Hyperledger Fabric --- Consensus and Transaction Flow}
\footnotesize
\lbl{Execute-Order-Validate (EOV) model:}
\begin{enumerate}\setlength\itemsep{2pt}
  \item \textbf{Execute:}
        Client submits transaction proposal to endorsing peers.
        Peers simulate chaincode execution, sign read/write set.
  \item \textbf{Order:}
        Client collects endorsements, submits to Ordering Service.
        Raft consensus produces a block containing the transaction.
  \item \textbf{Validate:}
        All peers validate endorsement policy and MVCC consistency.
        Valid transactions update the ledger state.
\end{enumerate}
\medskip
\lbl{D-IBFV Fabric configuration:}
\begin{itemize}
  \item Go chaincode: CID enrollment, lookup, revocation
  \item Endorsement policy: 1-of-1 (single org; extendable)
  \item Raft ordering: 3-node crash fault tolerant,
        0.5-second block timeout
  \item Channel isolation: D-IBFV transactions on dedicated channel
\end{itemize}
\medskip
\lbl{Why EOV matters for biometric systems:}
No global ordering required before execution
$\to$ parallel endorsement by multiple peers
$\to$ higher throughput than Ethereum's sequential execution.
\end{frame}
